node <<'NODE'
require('dotenv').config({quiet:true});
const fs = require('fs');
const {Client} = require('pg');
(async () => {
  const c = new Client({connectionString:process.env.DATABASE_URL});
  await c.connect();
  try {
    await c.query(fs.readFileSync('migrations/0046_preserve_booking_resources.sql','utf8'));
    const result = await c.query("SELECT conname, confdeltype, convalidated FROM pg_constraint WHERE conrelid='public.bookings'::regclass AND conname=ANY($1::text[])", [['bookings_master_id_masters_id_fk','bookings_service_id_services_id_fk']]);
    if(result.rows.length!==2 || result.rows.some(r=>r.confdeltype!=='r'||!r.convalidated)) throw new Error('Constraint verification failed');
    console.log('Both booking resource constraints validated: ON DELETE RESTRICT');
  } finally { await c.end(); }
})().catch(e=>{console.error('Migration failed:',e.code||e.message);process.exit(1)});
NODE
