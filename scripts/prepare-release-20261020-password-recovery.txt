set -euo pipefail
old=/var/www/aurelle-release-20261020-session-hardening
new=/var/www/aurelle-release-20261020-password-recovery
backup=/var/backups/aurelle/20261020-password-recovery
test ! -e "$new"
mkdir -p "$new" "$backup"
chmod 700 "$backup"
cp /etc/nginx/sites-available/aurelle "$backup/nginx.conf"
cp "$old/release.config.json" "$backup/previous.config.json"
cd "$old"
node <<'NODE'
require('dotenv').config({quiet:true});const {spawnSync}=require('child_process');const u=new URL(process.env.DATABASE_URL);const r=spawnSync('pg_dump',['-Fc','--no-owner','--no-acl','-f','/var/backups/aurelle/20261020-password-recovery/database.dump'],{env:{...process.env,PGHOST:u.hostname,PGPORT:u.port||'5432',PGUSER:decodeURIComponent(u.username),PGPASSWORD:decodeURIComponent(u.password),PGDATABASE:u.pathname.slice(1)},encoding:'utf8'});if(r.status!==0)throw new Error('Database backup failed');console.log('Database backup complete');
NODE
node <<'NODE'
require('dotenv').config({quiet:true});const {Client}=require('pg');(async()=>{const c=new Client({connectionString:process.env.DATABASE_URL});await c.connect();try{await c.query('BEGIN');await c.query("SET LOCAL lock_timeout='5s'");await c.query("ALTER TABLE users ADD COLUMN IF NOT EXISTS auth_version integer NOT NULL DEFAULT 0;\n-- Legacy reset tokens were stored as bearer credentials. Require a new link.\nUPDATE password_reset_tokens SET used_at = NOW() WHERE used_at IS NULL;\n");await c.query('COMMIT');console.log('Password recovery migration applied');}catch(e){await c.query('ROLLBACK');throw e;}finally{await c.end();}})().catch(()=>{console.error('Migration failed; deployment stopped');process.exit(1)});
NODE
tar -xzf /var/tmp/aurelle-20261020-password-recovery.tar.gz -C "$new"
cp "$old/.env" "$new/.env"
chmod 600 "$new/.env"
ln -s "$old/node_modules" "$new/node_modules"
ln -s /var/www/aurelle/attached_assets "$new/attached_assets"
mkdir -p "$new/server"
ln -s /var/www/aurelle/server/uploads "$new/server/uploads"
cp -an "$old/dist/public/assets/." "$new/dist/public/assets/"
node <<'NODE'
const fs=require('fs');const c=require('/var/www/aurelle-release-20261020-session-hardening/release.config.json');const a=c.apps[0];a.name='aurelle-production-20261020-password-recovery';a.cwd='/var/www/aurelle-release-20261020-password-recovery';a.script=a.cwd+'/dist/index.cjs';Object.assign(a.env,{PORT:'5022',APP_VERSION:'20261020-password-recovery',NODE_APP_INSTANCE:'1'});fs.writeFileSync(a.cwd+'/release.config.json',JSON.stringify(c,null,2),{mode:0o600});
NODE
cd "$new"
pm2 start "$new/release.config.json" >/dev/null
for i in $(seq 1 25); do if curl -fsS --max-time 3 http://127.0.0.1:5022/api/health > "$backup/new-health.json"; then break; fi; sleep 1; done
node -e "const h=require('$backup/new-health.json');if(h.status!=='healthy'||h.version!=='20261020-password-recovery')process.exit(1);console.log('New release healthy, database connected')"
